cmake_minimum_required(VERSION 3.16)
project(qingjian-fcitx5 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(QINGJIAN_RENDER_FFI "Link the Rust render-ffi static library" OFF)
option(QINGJIAN_EXPERIMENTAL_X11 "Enable unverified X11 prototype for explicit qingjian mode" OFF)
find_package(PkgConfig REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(FCITX5 REQUIRED IMPORTED_TARGET Fcitx5Core>=5.1.9)
set(QINGJIAN_PANEL_SOURCES src/panel/controller.cpp src/panel/backend/xcb.cpp src/panel/backend/screens.cpp src/panel/appearance.cpp)
set(QINGJIAN_SOURCES src/qingjian.cpp src/factory.cpp src/ipc/connection.cpp src/key/mapping.cpp ${QINGJIAN_PANEL_SOURCES})
if(QINGJIAN_RENDER_FFI)
  set(QINGJIAN_RENDER_FFI_LIB "" CACHE FILEPATH "Path to libqingjian_render_ffi.a")
  set(QINGJIAN_RENDER_FFI_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../render-ffi/include" CACHE PATH "Directory containing qingjian_render.h")
  if(NOT EXISTS "${QINGJIAN_RENDER_FFI_LIB}" OR NOT EXISTS "${QINGJIAN_RENDER_FFI_INCLUDE}/qingjian_render.h")
    message(FATAL_ERROR "Build qingjian-render-ffi first and set QINGJIAN_RENDER_FFI_LIB")
  endif()
endif()
if(QINGJIAN_EXPERIMENTAL_X11)
  if(NOT QINGJIAN_RENDER_FFI)
    message(FATAL_ERROR "The X11 prototype requires QINGJIAN_RENDER_FFI")
  endif()
  pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb xcb-render xcb-randr)
endif()
function(qingjian_target target)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(${target} PRIVATE PkgConfig::FCITX5 nlohmann_json::nlohmann_json Threads::Threads)
  if(QINGJIAN_RENDER_FFI)
    target_compile_definitions(${target} PRIVATE QJ_RENDER_FFI=1)
    target_include_directories(${target} PRIVATE "${QINGJIAN_RENDER_FFI_INCLUDE}")
    target_link_libraries(${target} PRIVATE "${QINGJIAN_RENDER_FFI_LIB}" ${CMAKE_DL_LIBS} m)
    # 静态 archive 的未用业务/网络对象不进入插件；Rust 内部符号不污染 Fcitx。
    target_link_options(${target} PRIVATE "LINKER:--gc-sections" "LINKER:--exclude-libs,ALL")
  endif()
  if(QINGJIAN_EXPERIMENTAL_X11)
    target_compile_definitions(${target} PRIVATE QJ_EXPERIMENTAL_X11=1)
    target_link_libraries(${target} PRIVATE PkgConfig::XCB)
  endif()
endfunction()
add_library(qingjian MODULE ${QINGJIAN_SOURCES})
qingjian_target(qingjian)
set_target_properties(qingjian PROPERTIES PREFIX "" OUTPUT_NAME "qingjian")
install(TARGETS qingjian LIBRARY DESTINATION lib/fcitx5)
install(FILES data/addon/qingjian.conf DESTINATION share/fcitx5/addon)
install(FILES data/inputmethod/qingjian.conf DESTINATION share/fcitx5/inputmethod)
install(FILES ../../../assets/icon/logo.png DESTINATION share/pixmaps RENAME qingjian.png)
