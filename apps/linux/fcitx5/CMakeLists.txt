cmake_minimum_required(VERSION 3.16)
project(qingjian-fcitx5 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(nlohmann_json REQUIRED)
pkg_check_modules(FCITX5 REQUIRED IMPORTED_TARGET Fcitx5Core>=5.1.9)
add_library(qingjian MODULE src/qingjian.cpp src/factory.cpp src/ipc/connection.cpp src/key/mapping.cpp)
target_include_directories(qingjian PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(qingjian PRIVATE PkgConfig::FCITX5 nlohmann_json::nlohmann_json)
target_compile_options(qingjian PRIVATE -Wall -Wextra -Werror)
set_target_properties(qingjian PROPERTIES PREFIX "" OUTPUT_NAME "qingjian")
install(TARGETS qingjian LIBRARY DESTINATION lib/fcitx5)
install(FILES data/addon/qingjian.conf DESTINATION share/fcitx5/addon)
install(FILES data/inputmethod/qingjian.conf DESTINATION share/fcitx5/inputmethod)
install(FILES ../../../assets/icon/logo.png DESTINATION share/pixmaps RENAME qingjian.png)

include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(qingjian-integration tests/integration.cpp src/qingjian.cpp src/ipc/connection.cpp src/key/mapping.cpp)
  target_include_directories(qingjian-integration PRIVATE src)
  target_compile_options(qingjian-integration PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(qingjian-integration PRIVATE PkgConfig::FCITX5 nlohmann_json::nlohmann_json Threads::Threads)
  add_test(NAME qingjian-integration COMMAND qingjian-integration)
  set_tests_properties(qingjian-integration PROPERTIES TIMEOUT 10)
  add_executable(qingjian-lifecycle tests/lifecycle.cpp src/qingjian.cpp src/ipc/connection.cpp src/key/mapping.cpp)
  target_include_directories(qingjian-lifecycle PRIVATE src)
  target_compile_options(qingjian-lifecycle PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(qingjian-lifecycle PRIVATE PkgConfig::FCITX5 nlohmann_json::nlohmann_json Threads::Threads)
  foreach(scenario focusout-panel focusout-client focusout-client-owned shift-unknown shift-sensitive shift-password deactivate-sensitive deactivate-password deactivate-disabled capability-deactivate)
    add_test(NAME qingjian-${scenario} COMMAND qingjian-lifecycle ${scenario})
    set_tests_properties(qingjian-${scenario} PROPERTIES TIMEOUT 10)
  endforeach()
  add_test(NAME qingjian-uninstall-preserves-data COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/../tests/uninstall.sh)
endif()
